import json, datetime
from pathlib import Path
agg_start = datetime.datetime(2026, 5, 8, 22, 26, 15, tzinfo=datetime.timezone.utc)
fired, settled = [], {}
for p in sorted(Path('state/staging/decisions').glob('*.jsonl')):
    if p.suffix == '.bak': continue
    for line in p.read_text().splitlines():
        if not line.strip(): continue
        r = json.loads(line)
        if r.get('kind') == 'fired': fired.append(r)
        elif r.get('kind') == 'settled': settled[r['decision_id']] = r
agg = [f for f in fired if f['decision_ts_ns'] >= int(agg_start.timestamp()*1e9)]
filled = sum(1 for f in agg if (settled.get(f['decision_id']) or {}).get('fill_at_settle'))
sett = sum(1 for f in agg if (settled.get(f['decision_id']) or {}).get('resolution'))
won = sum(1 for f in agg if (settled.get(f['decision_id']) or {}).get('pnl_usd', 0) > 0)
total = sum((settled.get(f['decision_id']) or {}).get('pnl_usd') or 0.0 for f in agg)
elapsed = (datetime.datetime.now(datetime.timezone.utc) - agg_start).total_seconds() / 3600
print(f'elapsed: {elapsed:.1f}h')
print(f'fires: {len(agg)} filled: {filled} settled: {sett} won: {won}')
print(f'WR: {100*won/max(sett,1):.1f}% PnL: \${total:+.2f}')
print(f'fire-rate proj: {len(agg)*24/max(elapsed,0.01):.0f}/d')
"
