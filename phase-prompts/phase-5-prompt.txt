TASK 5.3 — LATENCY CHECK  (the gate test for locked decision #2)

This is THE most important number in Phase 5. If Dublin doesn't deliver,
locked decision #2 was wrong and we redo the region.

  # 1. ICMP ping (rough estimate, indicative only)
  ping -c 50 ws-subscriptions-clob.polymarket.com

  # 2. Actual TCP RTT to the WS port (more meaningful than ICMP)
  for i in $(seq 1 20); do
    /usr/bin/time -f "%e" curl -s -o /dev/null \
        --connect-timeout 5 \
        https://ws-subscriptions-clob.polymarket.com 2>&1
  done

  # 3. WS handshake + first message — uses our existing test infrastructure
  cd /home/ops/polymarket-5m-bot
  uv run python -c "
import asyncio, time, websockets
async def main():
    samples = []
    for _ in range(20):
        t0 = time.perf_counter()
        async with websockets.connect(
            'wss://ws-subscriptions-clob.polymarket.com/ws/market'
        ) as ws:
            samples.append((time.perf_counter() - t0) * 1000)
        await asyncio.sleep(0.5)
    samples.sort()
    p50 = samples[len(samples)//2]
    p99 = samples[int(len(samples)*0.99)]
    print(f'p50={p50:.2f}ms  p99={p99:.2f}ms  max={max(samples):.2f}ms  n={len(samples)}')
asyncio.run(main())
"
